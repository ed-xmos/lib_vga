cmake_minimum_required(VERSION 3.21)
include($ENV{XMOS_CMAKE_PATH}/xcommon.cmake)

project(test_vga)

if(NOT DEFINED APP_HW_TARGET AND NOT BUILD_NATIVE)
    set(APP_HW_TARGET XK-EVK-XU316)
endif()

set(APP_INCLUDES src)

set(APP_COMMON_BUILD_FLAGS
    -O3
    -g
    -report
    -Wall
)

# Shortened frame so a simulation covers several whole frames quickly.
# test_vga.py reads these -D flags to configure the test bench (vga_tb.py).
set(TEST_TIMING_FLAGS
    -DVGA_H_ACTIVE=64
    -DVGA_H_FRONT=8
    -DVGA_H_SYNC=8
    -DVGA_H_BACK=8
    -DVGA_V_ACTIVE=32
    -DVGA_V_FRONT=8
    -DVGA_V_SYNC=8
    -DVGA_V_BACK=8
)

# One build per resolution / colour depth
set(APP_COMPILER_FLAGS_640_4bpp   ${APP_COMMON_BUILD_FLAGS} ${TEST_TIMING_FLAGS} -DVGA_RESOLUTION=VGA_640_480 -DVGA_RGB_PORT=XS1_PORT_4A)
set(APP_COMPILER_FLAGS_640_8bpp   ${APP_COMMON_BUILD_FLAGS} ${TEST_TIMING_FLAGS} -DVGA_RESOLUTION=VGA_640_480 -DVGA_RGB_PORT=XS1_PORT_8A)
set(APP_COMPILER_FLAGS_640_16bpp  ${APP_COMMON_BUILD_FLAGS} ${TEST_TIMING_FLAGS} -DVGA_RESOLUTION=VGA_640_480 -DVGA_RGB_PORT=XS1_PORT_16A)
set(APP_COMPILER_FLAGS_320_4bpp   ${APP_COMMON_BUILD_FLAGS} ${TEST_TIMING_FLAGS} -DVGA_RESOLUTION=VGA_320_240 -DVGA_RGB_PORT=XS1_PORT_4A)
set(APP_COMPILER_FLAGS_320_8bpp   ${APP_COMMON_BUILD_FLAGS} ${TEST_TIMING_FLAGS} -DVGA_RESOLUTION=VGA_320_240 -DVGA_RGB_PORT=XS1_PORT_8A)
set(APP_COMPILER_FLAGS_320_16bpp  ${APP_COMMON_BUILD_FLAGS} ${TEST_TIMING_FLAGS} -DVGA_RESOLUTION=VGA_320_240 -DVGA_RGB_PORT=XS1_PORT_16A)

# Stress configs: burn threads fill the tile so the controller runs at a reduced
# MIPS share while streaming pixels. VGA_STRESS is the total active thread count:
# 8 threads = the minimum guaranteed 600/8 = 75 MHz per thread, 5 threads = 120 MHz.
set(APP_COMPILER_FLAGS_640_4bpp_stress   ${APP_COMMON_BUILD_FLAGS} ${TEST_TIMING_FLAGS} -DVGA_RESOLUTION=VGA_640_480 -DVGA_RGB_PORT=XS1_PORT_4A  -DVGA_STRESS=8)
set(APP_COMPILER_FLAGS_640_8bpp_stress   ${APP_COMMON_BUILD_FLAGS} ${TEST_TIMING_FLAGS} -DVGA_RESOLUTION=VGA_640_480 -DVGA_RGB_PORT=XS1_PORT_8A  -DVGA_STRESS=8)
set(APP_COMPILER_FLAGS_640_16bpp_stress5 ${APP_COMMON_BUILD_FLAGS} ${TEST_TIMING_FLAGS} -DVGA_RESOLUTION=VGA_640_480 -DVGA_RGB_PORT=XS1_PORT_16A -DVGA_STRESS=5)
set(APP_COMPILER_FLAGS_320_4bpp_stress   ${APP_COMMON_BUILD_FLAGS} ${TEST_TIMING_FLAGS} -DVGA_RESOLUTION=VGA_320_240 -DVGA_RGB_PORT=XS1_PORT_4A  -DVGA_STRESS=8)
set(APP_COMPILER_FLAGS_320_8bpp_stress   ${APP_COMMON_BUILD_FLAGS} ${TEST_TIMING_FLAGS} -DVGA_RESOLUTION=VGA_320_240 -DVGA_RGB_PORT=XS1_PORT_8A  -DVGA_STRESS=8)
set(APP_COMPILER_FLAGS_320_16bpp_stress  ${APP_COMMON_BUILD_FLAGS} ${TEST_TIMING_FLAGS} -DVGA_RESOLUTION=VGA_320_240 -DVGA_RGB_PORT=XS1_PORT_16A -DVGA_STRESS=8)

# Full standard 640x480 @ 60 Hz timing (no shortened frame): exercises the real
# divider values and port timer wrap arithmetic. Simulated for ~2.5 real frames.
set(APP_COMPILER_FLAGS_640_8bpp_full ${APP_COMMON_BUILD_FLAGS} -DVGA_RESOLUTION=VGA_640_480 -DVGA_RGB_PORT=XS1_PORT_8A)

set(APP_DEPENDENT_MODULES "lib_vga")

set(XMOS_SANDBOX_DIR ${CMAKE_CURRENT_LIST_DIR}/../../..)

XMOS_REGISTER_APP()
